127template <typename T>
128void SetArgument(asIScriptContext* ctx, int argIndex, T&& arg)
129{
130 int r;
131 if constexpr (std::is_pointer_v<std::decay_t<T>>)
132 {
133 r = ctx->SetArgObject(argIndex, arg);
134 }
135 else if constexpr (std::is_integral_v<std::decay_t<T>>)
136 {
137 r = ctx->SetArgDWord(argIndex, static_cast<asDWORD>(arg));
138 }
139 else if constexpr (std::is_floating_point_v<std::decay_t<T>>)
140 {
141 r = ctx->SetArgFloat(argIndex, static_cast<float>(arg));
142 }
143 else
144 {
145 LOG("Not supported type.");
146 }
147}

180std::optional<returnType> BCCGAngelscriptInfluence::ExecuteMethod(ScriptFunction function, vaArgs&&... args) const
181{
182 if (m_apScriptModule == nullptr)

186
187 if (m_ScriptObj == nullptr)
188 {
189 return std::nullopt;

191
192 std::optional<returnType> result;

194 asIScriptFunction* decl_func = m_ScriptFunc[(int)function];

197 if (decl_func)
